```python
import tkinter as tk

# Function for each button press
def btn_press(value):
    entry_value = entry.get()
    if value == '=':
        try:
            result = str(eval(entry_value))
            entry.delete(0, tk.END)
            entry.insert(tk.END, result)
        except Exception:
            entry.delete(0, tk.END)
            entry.insert(tk.END, "Error")
            return
    elif value == 'C':
        entry.delete(0, tk.END)
    else:
        entry.insert(tk.END, value)

# Create the main window
root = tk.Tk()
root.title("Calculator")

# Entry widget to display input and output
entry = tk.Entry(root, width=45, borderwidth=1)
entry.grid(row=0, column=0, columnspan=3, padx=10, pady=10)

# Buttons
buttons = [
    ('7', 1, 0), ('8', 1, 1), ('9', 1, 2),
    ('4', 2, 0), ('5', 2, 1), ('6', 2, 2),
    ('1', 3, 0), ('2', 3, 1), ('3', 3, 2),
    ('0', 4, 1), ('+', 1, 3), ('-', 2, 3), 
    ('*', 3, 3), ('/', 4, 0),
    ('=', 4, 2), ('C', 4, 3)
]

for (btn_text, row, col) in buttons:
    if btn_text != '=' and btn_text != 'C':
        button = tk.Button(root, text=btn_text, width=10, command=lambda v=btn_text: btn_press(v))
    elif btn_text == '=':
        button = tk.Button(root, text=btn_text, width=10, command=lambda: btn_press('='))
    else:  # Clear button
        button = tk.Button(root, text=btn_text, width=10, command=lambda: btn_press('C'))

    button.grid(row=row, column=col)

# Run the application
root.mainloop()
```
